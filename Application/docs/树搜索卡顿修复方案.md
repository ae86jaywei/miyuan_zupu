# 树搜索卡顿问题修复方案

> 问题：宝塔式谱、牒记式谱、现代创新式搜索卡顿约5秒
> 原因：同步全表遍历 + 跳转时线性搜索

---

## 一、问题定位

### 1.1 搜索卡顿原因

| 位置 | 代码 | 问题 |
|------|------|------|
| `GenealogyService.search()` | 同步遍历所有人员 | 数万人 × 9字段 = 大量计算 |
| `TreeCanvas.runSearch()` | 同步遍历 `people` 数组 | 阻塞主线程 |
| `focusOnUid()` 牒记式 | 线性搜索 `diePeople` | 每次跳转都要遍历 |

### 1.2 跳转卡顿原因

| 谱式 | 跳转方式 | 问题 |
|------|----------|------|
| 宝塔式 | 组件内线性搜索行 | 行数多时卡顿 |
| 牒记式 | 线性搜索 `diePeople` | 每次都要遍历 |
| 现代式 | 组件内线性搜索行 | 行数多时卡顿 |
| 欧式/苏式 | 遍历缓存 boxes | 已优化，风险低 |
| 分页 | `book.pageOf(uid)` | 已优化，风险低 |

---

## 二、修复方案

### 方案 1：建立搜索索引（推荐）

**文件：** `Application/common/base/src/main/ets/model/PersonIndex.ets`

```typescript
/**
 * 人员索引：加速搜索和跳转
 * 在 GenealogyService 初始化时构建，人员变更时更新
 */

export class PersonIndex {
  // uid -> Person 映射
  private uidMap: Map<string, Person> = new Map();
  
  // 姓名 -> Person[] 映射（支持模糊搜索）
  private nameIndex: Map<string, Person[]> = new Map();
  
  // 拼音首字母索引（可选）
  private pinyinIndex: Map<string, Person[]> = new Map();
  
  // 跳转索引：uid -> 行索引（宝塔式/现代式）
  private rowMap: Map<string, number> = new Map();
  
  // 跳转索引：uid -> 页码（牒记式）
  private diePageMap: Map<string, number> = new Map();
  
  private built: boolean = false;
  
  /**
   * 构建索引
   */
  build(people: Person[]): void {
    this.uidMap.clear();
    this.nameIndex.clear();
    this.rowMap.clear();
    this.diePageMap.clear();
    
    for (let i: number = 0; i < people.length; i = i + 1) {
      const p: Person = people[i];
      
      // uid 索引
      this.uidMap.set(p.uid.toUpperCase(), p);
      
      // 姓名索引（按首字母分组）
      const firstChar: string = p.name.charAt(0).toUpperCase();
      if (!this.nameIndex.has(firstChar)) {
        this.nameIndex.set(firstChar, []);
      }
      this.nameIndex.get(firstChar)!.push(p);
    }
    
    this.built = true;
  }
  
  /**
   * 按 UID 快速查找
   */
  getByUid(uid: string): Person | null {
    return this.uidMap.get(uid.toUpperCase()) || null;
  }
  
  /**
   * 模糊搜索（O(1) 复杂度）
   */
  search(keyword: string): Person[] {
    const k: string = keyword.trim().toLowerCase();
    if (k.length === 0) {
      return [];
    }
    
    const out: Person[] = [];
    const keyUpper: string = k.toUpperCase();
    
    // 优先按首字母查找
    if (k.length === 1) {
      const hits: Person[] | undefined = this.nameIndex.get(keyUpper);
      if (hits !== undefined) {
        return hits.slice(0, 40);
      }
    }
    
    // 遍历所有人员（但只在搜索时使用）
    for (const p of this.uidMap.values()) {
      if (p.name.toLowerCase().includes(k) ||
          p.alias.toLowerCase().includes(k) ||
          p.uid.toUpperCase().includes(keyUpper) ||
          p.generationChar.toLowerCase().includes(k)) {
        out.push(p);
        if (out.length >= 40) {
          break;
        }
      }
    }
    
    return out;
  }
  
  /**
   * 设置跳转索引
   */
  setRowMapping(uid: string, row: number): void {
    this.rowMap.set(uid.toUpperCase(), row);
  }
  
  /**
   * 获取行索引
   */
  getRow(uid: string): number {
    return this.rowMap.get(uid.toUpperCase()) ?? -1;
  }
  
  /**
   * 设置牒记式页码
   */
  setDiePage(uid: string, page: number): void {
    this.diePageMap.set(uid.toUpperCase(), page);
  }
  
  /**
   * 获取牒记式页码
   */
  getDiePage(uid: string): number {
    return this.diePageMap.get(uid.toUpperCase()) ?? -1;
  }
}
```

### 方案 2：优化 GenealogyService.search()

**文件：** `Application/entry/src/main/ets/service/GenealogyService.ets`

```typescript
/**
 * 优化后的搜索方法
 * 使用预构建的索引加速搜索
 */

// 在类中添加索引成员
private index: PersonIndex = new PersonIndex();

/**
 * 构建索引（在数据加载后调用）
 */
buildIndex(): void {
  this.index.build(this.people);
  
  // 构建牒记式页码索引
  if (this.diePeople !== null) {
    for (let i: number = 0; i < this.diePeople.length; i = i + 1) {
      this.index.setDiePage(this.diePeople[i].uid, i);
    }
  }
}

/**
 * 模糊搜索：谱名 / 曾用名 / 唯一ID / 辈分字
 * 优化后：O(1) 或 O(n) 但 n 是索引大小
 */
search(keyword: string): Person[] {
  const k: string = keyword.trim().toLowerCase();
  if (k.length === 0) {
    return this.people;
  }
  
  // 使用索引搜索
  return this.index.search(k);
}

/**
 * 按 UID 快速查找
 */
getByUid(uid: string): Person | null {
  return this.index.getByUid(uid);
}
```

### 方案 3：优化 TreeCanvas.runSearch()

**文件：** `Application/entry/src/main/ets/features/TreeCanvas.ets`

```typescript
/**
 * 优化后的搜索方法
 * 使用异步搜索避免阻塞主线程
 */
private async runSearch(q: string): Promise<void> {
  this.searchQuery = q;
  const key: string = q.trim();
  if (key.length === 0) {
    this.searchResults = [];
    return;
  }
  
  // 异步搜索，避免阻塞主线程
  const results: Person[] = await new Promise<Person[]>((resolve) => {
    setTimeout(() => {
      const people: Person[] = SR.genealogy().getPeople();
      const out: Person[] = [];
      const keyUpper: string = key.toUpperCase();
      
      for (let i: number = 0; i < people.length; i = i + 1) {
        const p: Person = people[i];
        if (p.name.includes(key) || p.alias.includes(key) || 
            p.generationChar.includes(key) ||
            p.uid.toUpperCase().includes(keyUpper)) {
          out.push(p);
          if (out.length >= 40) {
            break;
          }
        }
      }
      
      resolve(out);
    }, 0); // 延迟到下一个事件循环
  });
  
  this.searchResults = results;
}
```

### 方案 4：优化 focusOnUid() 牒记式跳转

**文件：** `Application/entry/src/main/ets/features/TreeCanvas.ets`

```typescript
/**
 * 优化后的跳转方法
 * 使用索引加速跳转
 */
private async focusOnUid(uid: string): Promise<void> {
  const svc: GenealogyService = SR.genealogy();
  const target: string = uid.toUpperCase();
  
  // 牒记式：使用索引加速跳转
  if (this.style === TreeStyle.DIE) {
    if (this.diePeople.length === 0) {
      this.rebuildDieSource();
    }
    
    // 使用索引查找
    const page: number = svc.getDiePage(uid);
    if (page >= 0 && page < this.diePeople.length) {
      this.onFocusPerson(uid);
      this.dieScroller.scrollToIndex(page, true, ScrollAlign.CENTER);
      return;
    }
    
    // 索引未命中，回退到线性搜索
    for (let i: number = 0; i < this.diePeople.length; i = i + 1) {
      if (this.diePeople[i].uid.toUpperCase() === target) {
        // 更新索引
        svc.setDiePage(uid, i);
        this.onFocusPerson(uid);
        this.dieScroller.scrollToIndex(i, true, ScrollAlign.CENTER);
        return;
      }
    }
    
    this.getUIContext().getPromptAction().showToast({ message: T('未找到该族人') });
    return;
  }
  
  // ... 其他谱式保持不变
}
```

### 方案 5：优化宝塔式/现代式组件内跳转

**文件：** `Application/entry/src/main/ets/components/tree/PagodaTree.ets`

```typescript
/**
 * 优化后的跳转方法
 * 使用索引加速跳转
 */

// 添加跳转索引
private uidToRow: Map<string, number> = new Map();

onFocusChanged(): void {
  if (this.focusUid.length === 0) {
    return;
  }
  
  const target: string = this.focusUid.toUpperCase();
  
  // 使用索引查找
  const row: number = this.uidToRow.get(target) ?? -1;
  if (row >= 0 && row < this.rows.length) {
    this.scroller.scrollToIndex(row, true, ScrollAlign.CENTER);
    return;
  }
  
  // 索引未命中，构建索引并搜索
  this.buildUidIndex();
  
  const newRow: number = this.uidToRow.get(target) ?? -1;
  if (newRow >= 0 && newRow < this.rows.length) {
    this.scroller.scrollToIndex(newRow, true, ScrollAlign.CENTER);
    return;
  }
}

/**
 * 构建 UID -> 行索引映射
 */
private buildUidIndex(): void {
  this.uidToRow.clear();
  for (let i: number = 0; i < this.rows.length; i = i + 1) {
    const row: PagodaRow = this.rows[i];
    for (let k: number = row.start; k < row.start + row.count; k = k + 1) {
      this.uidToRow.set(row.nodes[k].uid.toUpperCase(), i);
    }
  }
}
```
