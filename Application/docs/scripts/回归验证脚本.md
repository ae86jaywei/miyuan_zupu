## 1. 字段一致性检查脚本

**文件：** `check-field-consistency.js`

**用途：** 检查 `FieldRegistry` 与数据库、云函数、CSV 的字段一致性

```javascript
#!/usr/bin/env node

/**
 * 字段一致性检查脚本
 * 检查 FieldRegistry 与以下位置的一致性：
 * 1. 数据库建表列（DatabaseSchema.ets）
 * 2. 云函数白名单（CloudProgram/cloudfunctions/_shared/patch.js）
 * 3. CSV 表头（ExportService.ets）
 */

const fs = require('fs');
const path = require('path');

// 颜色输出
const colors = {
  green: '\x1b[32m',
  red: '\x1b[31m',
  yellow: '\x1b[33m',
  blue: '\x1b[34m',
  reset: '\x1b[0m'
};

function log(color, msg) {
  console.log(`${colors[color]}${msg}${colors.reset}`);
}

// 解析 FieldRegistry
function parseFieldRegistry(filePath) {
  const content = fs.readFileSync(filePath, 'utf-8');
  const fields = [];
  
  // 匹配 PERSON_FIELDS 数组
  const arrayMatch = content.match(/PERSON_FIELDS:\s*FieldDef\[\]\s*=\s*\[([\s\S]*?)\];/);
  if (!arrayMatch) {
    log('red', '❌ 未找到 PERSON_FIELDS 定义');
    return null;
  }
  
  // 提取每个字段的 key
  const fieldMatches = arrayMatch[1].match(/\{[\s\S]*?key:\s*['"]([^'"]+)['"]/g);
  if (fieldMatches) {
    fieldMatches.forEach(match => {
      const keyMatch = match.match(/key:\s*['"]([^'"]+)['"]/);
      if (keyMatch) {
        fields.push(keyMatch[1]);
      }
    });
  }
  
  return fields;
}

// 解析数据库列
function parseDbColumns(filePath) {
  const content = fs.readFileSync(filePath, 'utf-8');
  const columns = [];
  
  // 匹配 COLUMNS 数组
  const arrayMatch = content.match(/COLUMNS:\s*string\[\]\s*=\s*\[([\s\S]*?)\];/);
  if (!arrayMatch) {
    log('red', '❌ 未找到 COLUMNS 定义');
    return null;
  }
  
  // 提取每个列名
  const columnMatches = arrayMatch[1].match(/['"]([^'"]+)['"]/g);
  if (columnMatches) {
    columnMatches.forEach(match => {
      columns.push(match.replace(/['"]/g, ''));
    });
  }
  
  return columns;
}

// 解析云函数白名单
function parseCloudWhitelist(filePath) {
  const content = fs.readFileSync(filePath, 'utf-8');
  const whitelist = [];
  
  // 匹配 PATCHABLE_FIELDS
  const match = content.match(/PATCHABLE_FIELDS\s*=\s*\[([\s\S]*?)\];/);
  if (match) {
    const fieldMatches = match[1].match(/['"]([^'"]+)['"]/g);
    if (fieldMatches) {
      fieldMatches.forEach(f => {
        whitelist.push(f.replace(/['"]/g, ''));
      });
    }
  }
  
  return whitelist;
}

// 解析 CSV 表头
function parseCsvHeaders(filePath) {
  const content = fs.readFileSync(filePath, 'utf-8');
  const headers = [];
  
  // 匹配 HEADERS 数组
  const arrayMatch = content.match(/HEADERS:\s*string\[\]\s*=\s*\[([\s\S]*?)\];/);
  if (!arrayMatch) {
    log('red', '❌ 未找到 HEADERS 定义');
    return null;
  }
  
  // 提取每个表头
  const headerMatches = arrayMatch[1].match(/['"]([^'"]+)['"]/g);
  if (headerMatches) {
    headerMatches.forEach(match => {
      headers.push(match.replace(/['"]/g, ''));
    });
  }
  
  return headers;
}

// 比较数组
function compareArrays(name1, arr1, name2, arr2) {
  const set1 = new Set(arr1);
  const set2 = new Set(arr2);
  
  const onlyIn1 = arr1.filter(x => !set2.has(x));
  const onlyIn2 = arr2.filter(x => !set1.has(x));
  
  if (onlyIn1.length > 0) {
    log('yellow', `⚠️  ${name1} 中有但 ${name2} 中没有：`);
    onlyIn1.forEach(f => log('yellow', `    - ${f}`));
  }
  
  if (onlyIn2.length > 0) {
    log('yellow', `⚠️  ${name2} 中有但 ${name1} 中没有：`);
    onlyIn2.forEach(f => log('yellow', `    - ${f}`));
  }
  
  return onlyIn1.length === 0 && onlyIn2.length === 0;
}

// 主函数
function main() {
  log('blue', '🔍 字段一致性检查...\n');
  
  const baseDir = process.cwd();
  
  // 文件路径
  const fieldRegistryPath = path.join(baseDir, 'Application/common/base/src/main/ets/model/FieldRegistry.ets');
  const dbSchemaPath = path.join(baseDir, 'Application/entry/src/main/ets/service/db/DatabaseSchema.ets');
  const cloudWhitelistPath = path.join(baseDir, 'Program/cloudfunctions/_shared/patch.js');
  const exportServicePath = path.join(baseDir, 'Application/entry/src/main/ets/service/ExportService.ets');
  
  // 检查文件是否存在
  if (!fs.existsSync(fieldRegistryPath)) {
    log('red', `❌ 文件不存在: ${fieldRegistryPath}`);
    process.exit(1);
  }
  
  // 解析 FieldRegistry
  log('blue', `📄 解析 FieldRegistry: ${fieldRegistryPath}`);
  const fieldRegistry = parseFieldRegistry(fieldRegistryPath);
  if (!fieldRegistry) process.exit(1);
  log('green', `   ✓ 找到 ${fieldRegistry.length} 个字段\n`);
  
  // 检查数据库列
  if (fs.existsSync(dbSchemaPath)) {
    log('blue', `📄 解析数据库列: ${dbSchemaPath}`);
    const dbColumns = parseDbColumns(dbSchemaPath);
    if (dbColumns) {
      log('green', `   ✓ 找到 ${dbColumns.length} 个列`);
      const match = compareArrays('FieldRegistry', fieldRegistry, '数据库列', dbColumns);
      if (match) log('green', '   ✓ 数据库列一致\n');
    }
  }
  
  // 检查云函数白名单
  if (fs.existsSync(cloudWhitelistPath)) {
    log('blue', `📄 解析云函数白名单: ${cloudWhitelistPath}`);
    const cloudWhitelist = parseCloudWhitelist(cloudWhitelistPath);
    if (cloudWhitelist) {
      log('green', `   ✓ 找到 ${cloudWhitelist.length} 个字段`);
      const match = compareArrays('FieldRegistry', fieldRegistry, '云函数白名单', cloudWhitelist);
      if (match) log('green', '   ✓ 云函数白名单一致\n');
    }
  }
  
  // 检查 CSV 表头
  if (fs.existsSync(exportServicePath)) {
    log('blue', `📄 解析 CSV 表头: ${exportServicePath}`);
    const csvHeaders = parseCsvHeaders(exportServicePath);
    if (csvHeaders) {
      log('green', `   ✓ 找到 ${csvHeaders.length} 个表头`);
      const match = compareArrays('FieldRegistry', fieldRegistry, 'CSV 表头', csvHeaders);
      if (match) log('green', '   ✓ CSV 表头一致\n');
    }
  }
  
  log('blue', '✅ 字段一致性检查完成');
}

main();
```

## 2. 编辑页覆盖检查脚本

**文件：** `check-person-edit-coverage.js`

**用途：** 检查 `PersonEdit.ets` 是否覆盖了所有 `FieldRegistry` 字段

```javascript
#!/usr/bin/env node

/**
 * 编辑页覆盖检查脚本
 * 检查 PersonEdit.ets 是否覆盖了 FieldRegistry 中的所有字段
 */

const fs = require('fs');
const path = require('path');

const colors = {
  green: '\x1b[32m',
  red: '\x1b[31m',
  yellow: '\x1b[33m',
  blue: '\x1b[34m',
  reset: '\x1b[0m'
};

function log(color, msg) {
  console.log(`${colors[color]}${msg}${colors.reset}`);
}

// 解析 FieldRegistry
function parseFieldRegistry(filePath) {
  const content = fs.readFileSync(filePath, 'utf-8');
  const fields = [];
  
  const arrayMatch = content.match(/PERSON_FIELDS:\s*FieldDef\[\]\s*=\s*\[([\s\S]*?)\];/);
  if (!arrayMatch) {
    log('red', '❌ 未找到 PERSON_FIELDS 定义');
    return null;
  }
  
  const fieldMatches = arrayMatch[1].match(/\{[\s\S]*?key:\s*['"]([^'"]+)['"]/g);
  if (fieldMatches) {
    fieldMatches.forEach(match => {
      const keyMatch = match.match(/key:\s*['"]([^'"]+)['"]/);
      if (keyMatch) {
        fields.push({
          key: keyMatch[1],
          type: match.match(/type:\s*['"]([^'"]+)['"]/)?.[1] || 'text'
        });
      }
    });
  }
  
  return fields;
}

// 解析 PersonEdit 字段
function parsePersonEditFields(filePath) {
  const content = fs.readFileSync(filePath, 'utf-8');
  const fields = new Set();
  
  // 匹配 valueOf 中的 switch case
  const valueOfMatch = content.match(/valueOf\(key:\s*string\):\s*string\s*\{([\s\S]*?)\}/);
  if (valueOfMatch) {
    const caseMatches = valueOfMatch[1].match(/case\s+['"]([^'"]+)['"]:/g);
    if (caseMatches) {
      caseMatches.forEach(c => {
        const keyMatch = c.match(/case\s+['"]([^'"]+)['"]:/);
        if (keyMatch) fields.add(keyMatch[1]);
      });
    }
  }
  
  // 匹配 assignValue 中的 switch case
  const assignValueMatch = content.match(/assignValue\(key:\s*string,\s*value:\s*string\):\s*void\s*\{([\s\S]*?)\}/);
  if (assignValueMatch) {
    const caseMatches = assignValueMatch[1].match(/case\s+['"]([^'"]+)['"]:/g);
    if (caseMatches) {
      caseMatches.forEach(c => {
        const keyMatch = c.match(/case\s+['"]([^'"]+)['"]:/);
        if (keyMatch) fields.add(keyMatch[1]);
      });
    }
  }
  
  // 匹配 formSnapshot 中的字段
  const snapshotMatch = content.match(/formSnapshot\(\):\s*Record<string,\s*string>\s*\{([\s\S]*?)\}/);
  if (snapshotMatch) {
    const fieldMatches = snapshotMatch[1].match(/['"]([^'"]+)['"]:/g);
    if (fieldMatches) {
      fieldMatches.forEach(f => {
        const keyMatch = f.match(/['"]([^'"]+)['"]:/);
        if (keyMatch) fields.add(keyMatch[1]);
      });
    }
  }
  
  return Array.from(fields);
}

// 主函数
function main() {
  log('blue', '🔍 编辑页覆盖检查...\n');
  
  const baseDir = process.cwd();
  
  const fieldRegistryPath = path.join(baseDir, 'Application/common/base/src/main/ets/model/FieldRegistry.ets');
  const personEditPath = path.join(baseDir, 'Application/entry/src/main/ets/pages/PersonEdit.ets');
  
  if (!fs.existsSync(fieldRegistryPath)) {
    log('red', `❌ 文件不存在: ${fieldRegistryPath}`);
    process.exit(1);
  }
  
  if (!fs.existsSync(personEditPath)) {
    log('red', `❌ 文件不存在: ${personEditPath}`);
    process.exit(1);
  }
  
  // 解析 FieldRegistry
  log('blue', `📄 解析 FieldRegistry: ${fieldRegistryPath}`);
  const fieldRegistry = parseFieldRegistry(fieldRegistryPath);
  if (!fieldRegistry) process.exit(1);
  log('green', `   ✓ 找到 ${fieldRegistry.length} 个字段\n`);
  
  // 解析 PersonEdit
  log('blue', `📄 解析 PersonEdit: ${personEditPath}`);
  const personEditFields = parsePersonEditFields(personEditPath);
  log('green', `   ✓ 找到 ${personEditFields.length} 个覆盖字段\n`);
  
  // 比较
  const registryKeys = fieldRegistry.map(f => f.key);
  const missing = registryKeys.filter(k => !personEditFields.includes(k));
  const extra = personEditFields.filter(k => !registryKeys.includes(k));
  
  if (missing.length > 0) {
    log('yellow', '⚠️  FieldRegistry 中有但 PersonEdit 未覆盖：');
    missing.forEach(k => {
      const field = fieldRegistry.find(f => f.key === k);
      log('yellow', `    - ${k} (${field?.type || 'unknown'})`);
    });
  }
  
  if (extra.length > 0) {
    log('yellow', '⚠️  PersonEdit 中有但 FieldRegistry 中没有：');
    extra.forEach(k => log('yellow', `    - ${k}`));
  }
  
  if (missing.length === 0 && extra.length === 0) {
    log('green', '✅ 编辑页覆盖检查通过，所有字段均已覆盖');
  } else {
    log('red', '❌ 编辑页覆盖检查失败');
    process.exit(1);
  }
}

main();
```

## 3. 云函数白名单检查脚本

**文件：** `check-cloud-whitelist.js`

**用途：** 检查云函数白名单与 FieldRegistry 的一致性

```javascript
#!/usr/bin/env node

/**
 * 云函数白名单检查脚本
 * 检查 CloudProgram/cloudfunctions/_shared/patch.js 中的 PATCHABLE_FIELDS 和 FIELD_LABELS
 */

const fs = require('fs');
const path = require('path');

const colors = {
  green: '\x1b[32m',
  red: '\x1b[31m',
  yellow: '\x1b[33m',
  blue: '\x1b[34m',
  reset: '\x1b[0m'
};

function log(color, msg) {
  console.log(`${colors[color]}${msg}${colors.reset}`);
}

// 解析 FieldRegistry
function parseFieldRegistry(filePath) {
  const content = fs.readFileSync(filePath, 'utf-8');
  const fields = [];
  
  const arrayMatch = content.match(/PERSON_FIELDS:\s*FieldDef\[\]\s*=\s*\[([\s\S]*?)\];/);
  if (!arrayMatch) {
    log('red', '❌ 未找到 PERSON_FIELDS 定义');
    return null;
  }
  
  const fieldMatches = arrayMatch[1].match(/\{[\s\S]*?key:\s*['"]([^'"]+)['"]/g);
  if (fieldMatches) {
    fieldMatches.forEach(match => {
      const keyMatch = match.match(/key:\s*['"]([^'"]+)['"]/);
      const labelMatch = match.match(/label:\s*['"]([^'"]+)['"]/);
      if (keyMatch) {
        fields.push({
          key: keyMatch[1],
          label: labelMatch?.[1] || ''
        });
      }
    });
  }
  
  return fields;
}

// 解析云函数白名单
function parseCloudWhitelist(filePath) {
  const content = fs.readFileSync(filePath, 'utf-8');
  const result = {
    fields: [],
    labels: {}
  };
  
  // 匹配 PATCHABLE_FIELDS
  const fieldsMatch = content.match(/PATCHABLE_FIELDS\s*=\s*\[([\s\S]*?)\];/);
  if (fieldsMatch) {
    const fieldMatches = fieldsMatch[1].match(/['"]([^'"]+)['"]/g);
    if (fieldMatches) {
      fieldMatches.forEach(f => {
        result.fields.push(f.replace(/['"]/g, ''));
      });
    }
  }
  
  // 匹配 FIELD_LABELS
  const labelsMatch = content.match(/FIELD_LABELS\s*=\s*\{([\s\S]*?)\};/);
  if (labelsMatch) {
    const labelMatches = labelsMatch[1].match(/['"]([^'"]+)['"]:\s*['"]([^'"]+)['"]/g);
    if (labelMatches) {
      labelMatches.forEach(l => {
        const match = l.match(/['"]([^'"]+)['"]:\s*['"]([^'"]+)['"]/);
        if (match) {
          result.labels[match[1]] = match[2];
        }
      });
    }
  }
  
  return result;
}

// 主函数
function main() {
  log('blue', '🔍 云函数白名单检查...\n');
  
  const baseDir = process.cwd();
  
  const fieldRegistryPath = path.join(baseDir, 'Application/common/base/src/main/ets/model/FieldRegistry.ets');
  const cloudWhitelistPath = path.join(baseDir, 'Program/cloudfunctions/_shared/patch.js');
  
  if (!fs.existsSync(fieldRegistryPath)) {
    log('red', `❌ 文件不存在: ${fieldRegistryPath}`);
    process.exit(1);
  }
  
  if (!fs.existsSync(cloudWhitelistPath)) {
    log('red', `❌ 文件不存在: ${cloudWhitelistPath}`);
    process.exit(1);
  }
  
  // 解析 FieldRegistry
  log('blue', `📄 解析 FieldRegistry: ${fieldRegistryPath}`);
  const fieldRegistry = parseFieldRegistry(fieldRegistryPath);
  if (!fieldRegistry) process.exit(1);
  log('green', `   ✓ 找到 ${fieldRegistry.length} 个字段\n`);
  
  // 解析云函数白名单
  log('blue', `📄 解析云函数白名单: ${cloudWhitelistPath}`);
  const cloudWhitelist = parseCloudWhitelist(cloudWhitelistPath);
  log('green', `   ✓ 找到 ${cloudWhitelist.fields.length} 个白名单字段`);
  log('green', `   ✓ 找到 ${Object.keys(cloudWhitelist.labels).length} 个中文名\n`);
  
  // 比较白名单
  const registryKeys = fieldRegistry.map(f => f.key);
  const missing = registryKeys.filter(k => !cloudWhitelist.fields.includes(k));
  const extra = cloudWhitelist.fields.filter(k => !registryKeys.includes(k));
  
  if (missing.length > 0) {
    log('yellow', '⚠️  FieldRegistry 中有但云函数白名单中没有：');
    missing.forEach(k => {
      const field = fieldRegistry.find(f => f.key === k);
      log('yellow', `    - ${k} (${field?.label})`);
    });
  }
  
  if (extra.length > 0) {
    log('yellow', '⚠️  云函数白名单中有但 FieldRegistry 中没有：');
    extra.forEach(k => log('yellow', `    - ${k}`));
  }
  
  // 检查中文名
  const missingLabels = fieldRegistry.filter(f => !cloudWhitelist.labels[f.key]);
  if (missingLabels.length > 0) {
    log('yellow', '⚠️  云函数中缺少中文名：');
    missingLabels.forEach(f => log('yellow', `    - ${f.key}: ${f.label}`));
  }
  
  if (missing.length === 0 && extra.length === 0 && missingLabels.length === 0) {
    log('green', '✅ 云函数白名单检查通过');
  } else {
    log('red', '❌ 云函数白名单检查失败');
    process.exit(1);
  }
}

main();
```

## 4. 使用说明

### 4.1 安装依赖

无需安装额外依赖，使用 Node.js 内置模块。

### 4.2 运行脚本

```bash
# 进入 docs 目录
cd Application/docs

# 运行字段一致性检查
node scripts/check-field-consistency.js

# 运行编辑页覆盖检查
node scripts/check-person-edit-coverage.js

# 运行云函数白名单检查
node scripts/check-cloud-whitelist.js

# 批量运行
node scripts/run-all.js
```

### 4.3 批量运行脚本

**文件：** `scripts/run-all.js`

```javascript
#!/usr/bin/env node

const { execSync } = require('child_process');
const path = require('path');

const scripts = [
  'check-field-consistency.js',
  'check-person-edit-coverage.js',
  'check-cloud-whitelist.js'
];

let failed = 0;

scripts.forEach(script => {
  console.log('\n' + '='.repeat(60));
  console.log(`运行: ${script}`);
  console.log('='.repeat(60));
  
  try {
    execSync(`node ${script}`, { stdio: 'inherit' });
  } catch (error) {
    failed++;
  }
});

console.log('\n' + '='.repeat(60));
if (failed === 0) {
  console.log('✅ 所有检查通过');
} else {
  console.log(`❌ ${failed} 个检查失败`);
  process.exit(1);
}
```
